/* 页脚模块：侧边栏「符文卷轴」入口 + 随机问题 + 外部链接。
   使用指南、符文占卜在页眉模块（ans-guide / ans-rune）。 */
window.AnsCustom = window.AnsCustom || {};
var NS = 'ans-custom';
var SIDEBAR_CLASS = NS + '-left-injected';
var EXTERNAL_URL = 'https://lusgli-0.github.io';
var POLL_MS = 400;
var POLL_MAX = 75;

if (window.AnsCustom.__ready) {
  window.AnsCustom.run();
} else {
  window.AnsCustom.__ready = true;

  function getBasePath() {
    var link = document.querySelector('link[href*="custom.css"]');
    if (!link) return '';
    var href = link.getAttribute('href') || '';
    var m = href.match(/^(.*)\/custom\.css(?:\?.*)?$/);
    return m ? m[1] : '';
  }

  function questionBase() {
    var a = document.querySelector('a[href*="/questions/"]');
    if (a) {
      var m = (a.getAttribute('href') || '').match(/^(.*)\/questions\/\d+/);
      if (m) return m[1];
    }
    return getBasePath();
  }

  function injectEntryStyle() {
    if (document.getElementById(NS + '-entry-style')) return;
    var s = document.createElement('style');
    s.id = NS + '-entry-style';
    s.textContent = '.' + NS + '-entry-title{color:#ffffff !important;letter-spacing:.08em;' +
      'text-shadow:0 0 14px rgba(139,92,246,.45);}';
    document.head.appendChild(s);
  }

  function findSideNavs() {
    var list = Array.prototype.slice.call(document.querySelectorAll('#sideNav'));
    if (list.length) return list;
    var fb = document.querySelector('#pcSideNav .nav.nav-pills, #mobileSideNav .nav.nav-pills');
    return fb ? [fb] : [];
  }

  function injectLeftSidebar() {
    injectEntryStyle();
    var navs = findSideNavs();
    if (!navs.length) return false;
    navs.forEach(function (nav) {
      if (nav.querySelector('.' + SIDEBAR_CLASS)) return;
      var w = document.createElement('div');
      w.className = SIDEBAR_CLASS;
      w.innerHTML =
        '<div class="py-2 px-3 mt-3 small fw-bold ' + NS + '-entry-title">符文卷轴</div>' +
        '<button type="button" class="nav-link border-0 bg-transparent text-start w-100 ' + NS + '-guide-trigger" style="cursor:pointer;">📖 社区公告</button>' +
        '<button type="button" class="nav-link border-0 bg-transparent text-start w-100 ' + NS + '-random-trigger" style="cursor:pointer;">🎲 随机问题</button>' +
        '<button type="button" class="nav-link border-0 bg-transparent text-start w-100 ' + NS + '-rune-trigger" style="cursor:pointer;">🔮 符文占卜</button>' +
        '<button type="button" class="nav-link border-0 bg-transparent text-start w-100 ' + NS + '-external-trigger" style="cursor:pointer;">🔗 外部链接</button>';
      nav.appendChild(w);
    });
    return true;
  }

  /* ===== 随机问题：多前缀尝试 + 页面链接兜底 + 站内跳转（不刷新） ===== */
  function apiCandidates() {
    var base = getBasePath(), out = [];
    [base, ''].forEach(function (b) {
      ['/answer/api/v1', '/api/v1'].forEach(function (p) {
        var u = b + p;
        if (out.indexOf(u) === -1) out.push(u);
      });
    });
    return out;
  }
  function tryRandomFrom(cands, idx, cb) {
    if (idx >= cands.length) { cb(null); return; }
    var api = cands[idx];
    fetch(api + '/question/page?page=1&page_size=20&order=newest')
      .then(function (r) { if (!r.ok) throw 0; return r.json(); })
      .then(function (d) {
        var data = (d && d.data) || {};
        var count = Number(data.count) || 0;
        if (!count) { tryRandomFrom(cands, idx + 1, cb); return; }
        var page = 1 + Math.floor(Math.random() * Math.max(1, Math.ceil(count / 20)));
        fetch(api + '/question/page?page=' + page + '&page_size=20&order=newest')
          .then(function (r2) { if (!r2.ok) throw 0; return r2.json(); })
          .then(function (d2) {
            var list = ((d2 && d2.data) || {}).list || [];
            if (!list.length) { tryRandomFrom(cands, idx + 1, cb); return; }
            cb(list[Math.floor(Math.random() * list.length)]);
          })
          .catch(function () { tryRandomFrom(cands, idx + 1, cb); });
      })
      .catch(function () { tryRandomFrom(cands, idx + 1, cb); });
  }
  function randomQuestionFromDom() {
    var links = Array.prototype.slice.call(document.querySelectorAll('a[href*="/questions/"]'))
      .filter(function (a) { return /\/questions\/\d+/.test(a.getAttribute('href') || ''); });
    return links.length ? links[Math.floor(Math.random() * links.length)].getAttribute('href') : null;
  }
  /* 站内跳转：pushState + popstate，交给 React Router 原地切换，不整页刷新 */
  function navTo(url) {
    if (!url) return;
    var cur = window.location.pathname + window.location.search + window.location.hash;
    if (cur === url) return;
    window.history.pushState({}, '', url);
    window.dispatchEvent(new PopStateEvent('popstate'));
  }
  function goRandomQuestion() {
    tryRandomFrom(apiCandidates(), 0, function (q) {
      if (q && q.id) {
        var url = questionBase() + '/questions/' + q.id;
        if (q.url_title) url += '/' + q.url_title;
        navTo(url);
        return;
      }
      navTo(randomQuestionFromDom());
    });
  }

  if (!window.AnsCustom.__bound) {
    window.AnsCustom.__bound = true;
    document.addEventListener('click', function (e) {
      if (e.target.closest('.' + NS + '-random-trigger')) {
        e.preventDefault(); goRandomQuestion(); return;
      }
      if (e.target.closest('.' + NS + '-external-trigger')) {
        e.preventDefault(); window.open(EXTERNAL_URL, '_blank', 'noopener,noreferrer'); return;
      }
    });
  }

  window.AnsCustom.run = function () {
    injectLeftSidebar();
  };
  window.AnsCustom.run();

  var pollCount = 0;
  var pollTimer = setInterval(function () {
    pollCount += 1;
    window.AnsCustom.run();
    if (pollCount >= POLL_MAX) clearInterval(pollTimer);
  }, POLL_MS);

  if (!window.AnsCustom.__observer) {
    window.AnsCustom.__observer = new MutationObserver(function () { window.AnsCustom.run(); });
    window.AnsCustom.__observer.observe(document.body, { childList: true, subtree: true });
  }
}

(function () {
  if (window.__ansParticleBg) return;
  window.__ansParticleBg = true;

  var COUNT_PER_AREA = 90 / (1600 * 900);   // 粒子密度（按面积缩放）
  var CONN_DIST = 140;
  var MOUSE_DIST = 200;
  var LINE_COLOR = '170,204,255';

  function injectCss() {
    var st = document.createElement('style');
    st.id = 'ansParticleBgStyle';
    st.textContent = [
      // 深色底：让粒子看得见；想更暗可加深颜色

      'body{background:transparent !important;}',
      // 粒子画布：垫底、全屏、不挡鼠标
      '#ans-particles-canvas{position:fixed;top:0;left:0;width:100vw;height:100vh;',
      'z-index:0;pointer-events:none;display:block;}',
      // 开关按钮样式
      '#ans-particles-btn{position:fixed;right:16px;bottom:16px;z-index:99999;',
      'padding:8px 16px;font-size:13px;line-height:1;color:#dfe7ff;',

    ].join('');
    document.head.appendChild(st);
  }

  function make() {
    injectCss();

    var canvas = document.createElement('canvas');
    canvas.id = 'ans-particles-canvas';
    document.body.appendChild(canvas);
    var ctx = canvas.getContext('2d');

    var W = 0, H = 0, dpr = Math.min(2, window.devicePixelRatio || 1);
    var mouse = { x: -9999, y: -9999 };
    var parts = [];

    // ---- 开关状态 ----
    var running = true;
    try {
      if (localStorage.getItem('ansParticleBg') === '0') running = false;
    } catch (e) {}

    // ---- 创建开关按钮 ----
    var btn = document.createElement('button');
    btn.id = 'ans-particles-btn';
    btn.type = 'button';
    btn.textContent = running ? '关闭动态背景' : '开启动态背景';
    if (!running) btn.classList.add('off');
    btn.addEventListener('click', function (e) {
      e.preventDefault();
      running = !running;
      canvas.style.display = running ? 'block' : 'none';
      btn.textContent = running ? '关闭动态背景' : '开启动态背景';
      btn.classList.toggle('off', !running);
      try { localStorage.setItem('ansParticleBg', running ? '1' : '0'); } catch (err) {}
      if (running) step(); // 重新启动动画循环
    });
    document.body.appendChild(btn);

    if (!running) canvas.style.display = 'none';

    function resize() {
      W = window.innerWidth;
      H = window.innerHeight;
      canvas.width = W * dpr;
      canvas.height = H * dpr;
      canvas.style.width = W + 'px';
      canvas.style.height = H + 'px';
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

      var target = Math.round(COUNT_PER_AREA * W * H);
      if (target > 160) target = 160;
      if (target < 40) target = 40;
      while (parts.length < target) parts.push(makeParticle(true));
      parts.length = target;
    }

    function makeParticle(anywhere) {
      var r = Math.random();
      var color = r < 0.45 ? [110,198,255] : r < 0.8 ? [179,136,255] : [255,213,79];
      return {
        x: anywhere ? Math.random() * (W || 1000) : Math.random() * W,
        y: anywhere ? Math.random() * (H || 1000) : Math.random() * H,
        vx: (Math.random() - 0.5) * 0.7,
        vy: (Math.random() - 0.5) * 0.7,
        baseR: Math.random() * 2.5 + 1,
        r: 1,
        baseA: Math.random() * 0.4 + 0.25,
        a: 0.4,
        color: color,
      };
    }

    function step() {
      if (!running) return; // 已关闭：停止绘制

      ctx.clearRect(0, 0, W, H);

      var i, j, p, q;
      for (i = 0; i < parts.length; i++) {
        p = parts[i];

        p.x += p.vx; p.y += p.vy;
        if (p.x < -20) p.x = W + 20; else if (p.x > W + 20) p.x = -20;
        if (p.y < -20) p.y = H + 20; else if (p.y > H + 20) p.y = -20;

        var dx = mouse.x - p.x, dy = mouse.y - p.y;
        var dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < MOUSE_DIST) {
          var force = (MOUSE_DIST - dist) / MOUSE_DIST;
          var ang = Math.atan2(dy, dx);
          p.vx -= Math.cos(ang) * force * 0.04;
          p.vy -= Math.sin(ang) * force * 0.04;
          p.r += (p.baseR + force * 3 - p.r) * 0.1;
          p.a += (Math.min(1, p.baseA + force * 0.5) - p.a) * 0.1;
        } else {
          p.r += (p.baseR - p.r) * 0.08;
          p.a += (p.baseA - p.a) * 0.08;
        }

        p.vx *= 0.999; p.vy *= 0.999;
        var spd = Math.sqrt(p.vx * p.vx + p.vy * p.vy);
        if (spd < 0.2) {
          var a0 = Math.random() * Math.PI * 2;
          p.vx += Math.cos(a0) * 0.3;
          p.vy += Math.sin(a0) * 0.3;
        } else if (spd > 1.8) {
          p.vx = p.vx / spd * 1.8;
          p.vy = p.vy / spd * 1.8;
        }

        // 连线
        for (j = i + 1; j < parts.length; j++) {
          q = parts[j];
          var lx = p.x - q.x, ly = p.y - q.y;
          var d2 = Math.sqrt(lx * lx + ly * ly);
          if (d2 < CONN_DIST) {
            ctx.strokeStyle = 'rgba(' + LINE_COLOR + ',' + ((1 - d2 / CONN_DIST) * 0.18).toFixed(3) + ')';
            ctx.lineWidth = 0.6;
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(q.x, q.y);
            ctx.stroke();
          }
        }

        // 粒子本体
        ctx.save();
        ctx.globalAlpha = p.a;
        ctx.fillStyle = 'rgb(' + p.color[0] + ',' + p.color[1] + ',' + p.color[2] + ')';
        ctx.shadowColor = 'rgba(' + p.color[0] + ',' + p.color[1] + ',' + p.color[2] + ',0.7)';
        ctx.shadowBlur = p.r * 6;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, 6.2832);
        ctx.fill();
        ctx.restore();
      }

      requestAnimationFrame(step);
    }

    window.addEventListener('resize', resize);
    window.addEventListener('mousemove', function (e) { mouse.x = e.clientX; mouse.y = e.clientY; });
    window.addEventListener('mouseout', function (e) {
      if (!e.relatedTarget) { mouse.x = -9999; mouse.y = -9999; }
    });
    window.addEventListener('touchmove', function (e) {
      if (e.touches.length) { mouse.x = e.touches[0].clientX; mouse.y = e.touches[0].clientY; }
    }, { passive: true });
    window.addEventListener('touchend', function () { mouse.x = -9999; mouse.y = -9999; });

    resize();
    if (running) step();
    console.log('[ParticleBg] 粒子背景已启动（密度自适应 / 鼠标排斥 / 连线 / 不挡点击 / 带开关按钮）');
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', make);
  } else {
    make();
  }
})();

        big: big,
        hit: 0,
        phase: rand(0, Math.PI * 2),

        spDir: Math.random() < 0.5 ? -1 : 1,
        orbR: rand(7, 24),
        col: big ? BIG_COL[(Math.random() * BIG_COL.length) | 0] : FINE_COL[(Math.random() * FINE_COL.length) | 0],
      });
    }
  }

  function drawSpark(ctx, mx, my, tm, al) {
    var jx = 1.68 * Math.sin(tm * 21) + 1.2 * Math.sin(tm * 13.7 + 2);
    var jy = 1.68 * Math.cos(tm * 17) + 1.2 * Math.sin(tm * 11.3 + 1);
    var x = mx + jx, y = my + jy;
    var fl = 0.72 + 0.28 * Math.sin(tm * 47);

    ctx.globalAlpha = al * fl;
    ctx.fillStyle = '#fffdf2';
    ctx.shadowColor = '#ffd54a';
    ctx.shadowBlur = 14 * al;
    ctx.beginPath();
    ctx.arc(x, y, 3.1, 0, 6.2832);
    ctx.fill();
    ctx.shadowBlur = 0;

    ctx.strokeStyle = 'rgba(255,248,220,' + (al * 0.95).toFixed(2) + ')';
    ctx.lineWidth = 0.9;
    for (var j = 0; j < 7; j++) {
      var ba = j * 2.399 + Math.sin(tm * 4.7 + j * 1.7) * 1.1 + Math.sin(tm * 9.3 + j * 0.9) * 0.4;
      var bl = 3.1 + Math.sin(tm * 7.1 + j * 2.3) * 1.44;
      ctx.beginPath();
      ctx.moveTo(x, y);
      ctx.lineTo(x + Math.cos(ba) * bl, y + Math.sin(ba) * bl);
      ctx.stroke();
    }
    ctx.globalAlpha = 1;
  }

  function clampToBall(mx, my) {

    if (d > lim) {

  // ================= 物理：紧绷极限弹簧 =================
  function physicsStep(h) {
    var d = hypot(posX, posY);
    var over = d - LIMIT_R;
    var ax, ay;

    if (dragging) {
      ax = SPRING_K * (targetX - posX) - SPRING_C * velX;
      ay = SPRING_K * (targetY - posY) - SPRING_C * velY;
    } else {
      ax = -SPRING_K * posX - SPRING_C * velX + Math.sin(time * 0.9) * 3 + rand(-1, 1) * 8;
      ay = -SPRING_K * posY - SPRING_C * velY + Math.sin(time * 1.4) * 2 + rand(-1, 1) * 8;
    }

    if (over > 0) {
      var prog = Math.min(1, over / LIMIT_SOFT);
      var kLim = LIMIT_K * (0.3 + 0.7 * prog);
      var nx = d > 0.001 ? posX / d : 1;
      var ny = d > 0.001 ? posY / d : 0;
      ax += -kLim * over * nx;
      ay += -kLim * over * ny;

      var vOut = velX * nx + velY * ny;
      if (vOut > 0) {
        var cLim = LIMIT_C * (0.4 + 0.6 * prog);
        ax += -cLim * vOut * nx;
        ay += -cLim * vOut * ny;
      }
    }

    ballAccX = ax;
    ballAccY = ay;

    velX += ax * h;
    velY += ay * h;
    posX += velX * h;
    posY += velY * h;

    var SAFE = LIMIT_R + LIMIT_SOFT + 10;
    var d2 = hypot(posX, posY);
    if (d2 > SAFE) {
      var sx = posX / d2, sy = posY / d2;
      posX = sx * SAFE;
      posY = sy * SAFE;
      var vn2 = velX * sx + velY * sy;
      if (vn2 > 0) {
        velX -= 1.5 * vn2 * sx;
        velY -= 1.5 * vn2 * sy;
      }
    }
  }

  // ================= 主循环 =================
  function start() {
    if (rafOn) return;
    rafOn = true; lastT = 0;
    requestAnimationFrame(loop);
  }
  function loop(now) {
    if (!el.stage || !doc.body.contains(el.stage)) { rafOn = false; return; }
    var dt = lastT ? (now - lastT) / 1000 : 0.016;
    lastT = now;
    if (dt > 0.05) dt = 0.05;
    step(dt);
    requestAnimationFrame(loop);
  }

  function step(dt) {
    time += dt;

    sub += dt;
    if (sub > 0.1) sub = 0.1;
    while (sub >= H) {
      physicsStep(H);
      sub -= H;
    }

    var spd = hypot(velX, velY);
    energy += (clamp(spd / 320, 0, 1) - energy) * Math.min(1, dt * 9);

    el.wrap.style.transform =
      'translate(' + posX.toFixed(2) + 'px,' + posY.toFixed(2) + 'px)' +
      ' rotate(' + (posX * 0.0028).toFixed(4) + 'rad)';
    var g = 0.16 + 0.74 * energy + (mouseOn ? 0.15 : 0);
    el.glow.style.opacity = g.toFixed(2);
    el.wrap.style.filter = 'drop-shadow(0 0 ' + (6 + 18 * energy).toFixed(0) + 'px rgba(150,170,255,' + (0.3 + 0.4 * energy).toFixed(2) + '))';
    el.shadow.style.opacity = (0.4 + 0.4 * energy).toFixed(2);
    el.shadow.style.transform = 'translateX(' + (posX * 0.55).toFixed(1) + 'px)';

    var ctx = el.ctx;
    ctx.clearRect(0, 0, CV, CV);

    ctx.save();
    ctx.beginPath();
    ctx.arc(OFF + R, OFF + R, R - 0.5, 0, 6.2832);
    ctx.clip();

    var mBallX = lastMouse ? lastMouse.x - OFF : R;
    var mBallY = lastMouse ? lastMouse.y - OFF : R;
    var lim = R - 3;

    for (var i = 0; i < particles.length; i++) {
      var p = particles[i];
      if (p.hit > 0) p.hit -= dt;

      var rx = p.x - R, ry = p.y - R, rr = hypot(rx, ry) || 1;

      p.vx -= ballAccX * INERTIA * dt;
      p.vy -= ballAccY * INERTIA * dt;

      if (mouseOn) {
        var mxb = mouseOn.x - OFF, myb = mouseOn.y - OFF;
        var relX = p.x - mxb, relY = p.y - myb;
        var dd = hypot(relX, relY) || 1;
        var a0 = Math.atan2(relY, relX);
        var speed = 55 + p.sp * 8;
        var radialErr = dd - p.orbR;
        var desVX = -Math.sin(a0) * speed * p.spDir - (relX / dd) * radialErr * 6;
        var desVY =  Math.cos(a0) * speed * p.spDir - (relY / dd) * radialErr * 6;
        var kb = Math.min(1, dt * 4.5);
        p.vx += (desVX - p.vx) * kb;
        p.vy += (desVY - p.vy) * kb;
        p.vx += rand(-1, 1) * 140 * dt;
        p.vy += rand(-1, 1) * 140 * dt;
        if (Math.random() < dt * 2.2) p.orbR = rand(7, 24);
      } else {
        var flow = 7 + p.sp * 1.6;
        var k2 = Math.min(1, dt * 1.8);
        p.vx += (-ry / rr * flow - p.vx) * k2;
        p.vy += ( rx / rr * flow - p.vy) * k2;

        var distM = hypot(p.x - mBallX, p.y - mBallY);
        var scatter = clamp(distM / R, 0.12, 1);
        var jitter = 12 + scatter * 105 + energy * 200;
        p.vx += rand(-1, 1) * jitter * dt;
        p.vy += rand(-1, 1) * jitter * dt;
      }

      var fd = Math.max(0, 1 - 1.5 * dt);
      p.vx *= fd; p.vy *= fd;
      p.x += p.vx * dt; p.y += p.vy * dt;

      rx = p.x - R; ry = p.y - R;
      var rr2 = hypot(rx, ry);
      if (rr2 > lim) {
        var nx2 = rx / rr2, ny2 = ry / rr2;
        p.x = R + nx2 * lim;
        p.y = R + ny2 * lim;
        var vn = p.vx * nx2 + p.vy * ny2;
        if (vn > 0) {
          p.vx -= (1 + REST) * vn * nx2;
          p.vy -= (1 + REST) * vn * ny2;
          var tj = rand(-45, 45);
          p.vx += -ny2 * tj + nx2 * rand(6, 18);
          p.vy +=  nx2 * tj + ny2 * rand(6, 18);
          p.hit = 0.4;
        }
      }

      var tw = 0.55 + 0.45 * Math.sin(time * p.sp + p.phase);
      var hitBoost = p.hit > 0 ? 1.4 : 1;
      ctx.globalAlpha = Math.min(1, tw * (p.big ? 0.95 : 0.7) * hitBoost);
      if (p.big) {
        ctx.fillStyle = p.col;
        ctx.shadowColor = '#ffe066';
        ctx.shadowBlur = p.hit > 0 ? 12 : 6;
        ctx.beginPath();
        ctx.arc(p.x + OFF, p.y + OFF, p.hit > 0 ? p.size * 1.3 : p.size, 0, 6.2832);
        ctx.fill();
        ctx.shadowBlur = 0;
      } else {
        ctx.fillStyle = p.col;
        if (p.hit > 0) {
          ctx.fillRect(p.x + OFF - 1, p.y + OFF - 1, 2, 2);
        } else {
          ctx.fillRect(p.x + OFF - 0.55, p.y + OFF - 0.55, 1.1, 1.1);
        }
      }
    }
    ctx.globalAlpha = 1;

    var cauX = R + posX * 0.08, cauY = R + 8 - posY * 0.05;
    var cg = ctx.createRadialGradient(cauX + OFF, cauY + OFF, 2, cauX + OFF, cauY + OFF, 36);
    var ca = 0.08 + 0.24 * energy;
    cg.addColorStop(0, 'rgba(255,244,190,' + ca.toFixed(2) + ')');
    cg.addColorStop(1, 'rgba(255,244,190,0)');
    ctx.fillStyle = cg;
    ctx.beginPath(); ctx.arc(cauX + OFF, cauY + OFF, 36, 0, 6.2832); ctx.fill();

    ctx.restore();

    if (mouseOn) {
      ctx.save();
      ctx.globalCompositeOperation = 'lighter';
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';
      drawSpark(ctx, mouseOn.x, mouseOn.y, time, 1);
      ctx.restore();
    }
  }

  mount();
  new MutationObserver(mount).observe(doc.body, { childList: true, subtree: true });
})();

(function () {
  if (window.__ansQuoteTyper) return;
  window.__ansQuoteTyper = true;

  var TYPE_MS = 85;    // 打字：每字间隔
  var DEL_MS  = 40;    // 回删：每字间隔
  var HOLD_MS = 2400;  // 打完停留（光标持续闪烁）
  var GAP_MS  = 250;   // 删空后到下一句的间隙

  function start(box) {
    if (!box || box.__typerOn) return;
    var spans = box.querySelectorAll('span');
    if (!spans.length) return;

    var lines = [];
    for (var i = 0; i < spans.length; i++) {
      var t = (spans[i].textContent || '').trim();
      if (t) lines.push(t);
    }
    if (!lines.length) return;

    box.__typerOn = true;
    box.innerHTML = '<span class="cq-text"></span><span class="cq-caret"></span>';
    var txt = box.querySelector('.cq-text');
    var caret = box.querySelector('.cq-caret');
    var idx = 0, pos = 0;

    function typeLine() {
      pos = 0;
      txt.textContent = '';
      caret.classList.add('cq-on');

      var iv = setInterval(function () {
        pos++;
        txt.textContent = lines[idx].slice(0, pos);
        if (pos >= lines[idx].length) {
          clearInterval(iv);
          setTimeout(deleteLine, HOLD_MS);
        }
      }, TYPE_MS);
    }

    function deleteLine() {
      var iv = setInterval(function () {
        pos--;
        txt.textContent = lines[idx].slice(0, pos);
        if (pos <= 0) {
          clearInterval(iv);
          caret.classList.remove('cq-on');
          setTimeout(function () {
            idx = (idx + 1) % lines.length;
            typeLine();
          }, GAP_MS);
        }
      }, DEL_MS);
    }

    typeLine();
    console.log('[QuoteTyper] 打字机已启动（打字 → 光标闪烁 → 回删 → 循环）');
  }

  function scan() {
    var boxes = document.querySelectorAll('.ans-crystal-quote');
    for (var i = 0; i < boxes.length; i++) {
      var b = boxes[i];
      if (!b.querySelector('.cq-text')) b.__typerOn = false;  // 被重建则重启
      start(b);
    }
  }

  scan();
  document.addEventListener('DOMContentLoaded', scan);
  new MutationObserver(scan).observe(document.body, { childList: true, subtree: true });
})();
