  /* 性能版倾斜：移动中关过渡直接跟手，移出才回正动画 */
  function bindTilt(panel) {
    if (!panel) return;
    var moving = false;
    panel.addEventListener('mousemove', function (e) {
      var r = panel.getBoundingClientRect();
      if (!r.width || !r.height) return;
      var px = (e.clientX - r.left) / r.width - .5;
      var py = (e.clientY - r.top) / r.height - .5;
      if (!moving) {
        moving = true;
        panel.style.transition = 'none';
      }
      panel.style.transform = 'rotateX(' + (-py * 8).toFixed(2) + 'deg) rotateY(' + (px * 8).toFixed(2) + 'deg)';
    });
    panel.addEventListener('mouseleave', function () {
      moving = false;
      panel.style.transition = 'transform .5s cubic-bezier(.23,1,.32,1)';
      panel.style.transform = 'rotateX(0deg) rotateY(0deg)';
    });
  }

  function bindGreenFX(panel) {
    var wrap = panel.querySelector('.ans-guide-particles');
    if (!wrap) return null;
    var arr = [], raf = null, last = 0, hideTimer = null;
    function spawn() {
      if (arr.length) return;
      var r = panel.getBoundingClientRect(), W = r.width || 800, H = r.height || 520, i;
      for (i = 0; i < 8; i++) {   /* 光点减到 8 个，进一步减压 */
        var d = document.createElement('span');
        d.className = 'ans-guide-dot';
        d._x = Math.random() * W; d._y = Math.random() * H;
        d._vx = (Math.random() - .5) * 36; d._vy = (Math.random() - .5) * 36;
        d.style.left = d._x + 'px'; d.style.top = d._y + 'px';
        wrap.appendChild(d); arr.push(d);
      }
      for (i = 0; i < 2; i++) {   /* 光斑减到 2 个 */
        var o = document.createElement('span');
        o.className = 'ans-guide-orb';
        var size = 100 + Math.random() * 90;
        o.style.width = size + 'px'; o.style.height = size + 'px';
        o._x = Math.random() * W; o._y = Math.random() * H;
        o._vx = (Math.random() - .5) * 22; o._vy = (Math.random() - .5) * 22;
        o.style.left = o._x + 'px'; o.style.top = o._y + 'px';
        wrap.appendChild(o); arr.push(o);
      }
    }
    function step(now) {
      var dt = Math.min((now - last) / 1000, .05); last = now;
      var r = panel.getBoundingClientRect(), W = r.width, H = r.height;
      if (!W || !H) { raf = null; return; }
      for (var i = 0; i < arr.length; i++) {
        var el = arr[i], m = 16;
        el._x += el._vx * dt; el._y += el._vy * dt;
        if (el._x < -m) el._x = W + m; else if (el._x > W + m) el._x = -m;
        if (el._y < -m) el._y = H + m; else if (el._y > H + m) el._y = -m;
        el.style.left = el._x + 'px'; el.style.top = el._y + 'px';
      }
      raf = requestAnimationFrame(step);
    }
    function show() {
      if (hideTimer) { clearTimeout(hideTimer); hideTimer = null; }
      spawn();
      void panel.offsetWidth;
      arr.forEach(function (el) { el.style.opacity = el.className.indexOf('orb') > -1 ? '.7' : '.9'; });
      if (!raf) { last = performance.now(); raf = requestAnimationFrame(step); }
    }
    function hide() {
      if (raf) { cancelAnimationFrame(raf); raf = null; }
      arr.forEach(function (el) { el.style.opacity = '0'; });
      var old = arr; arr = [];
      hideTimer = setTimeout(function () {
        old.forEach(function (el) { if (el.parentNode) el.parentNode.removeChild(el); });
        hideTimer = null;
      }, 650);
    }
    panel.addEventListener('mouseenter', show);
    panel.addEventListener('mouseleave', hide);
    return hide;
  }

  function startCountdown() {
    if (timer) clearInterval(timer);
    var el = els.rune.querySelector('.ans-rune-countdown');
    if (!el) return;
    function tick() {
      var now = new Date();
      var mid = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1);
      var s = Math.max(0, Math.floor((mid - now) / 1000));
      el.textContent = '距离下次占卜：' + pad2(Math.floor(s / 3600)) + ':' +
        pad2(Math.floor(s % 3600 / 60)) + ':' + pad2(s % 60);
    }
    tick();
    timer = setInterval(tick, 1000);
  }

  function drawRune() {
    var today = todayKey(), idx = null;
    try {
      var raw = JSON.parse(localStorage.getItem(KEY) || 'null');
      if (raw && raw.date === today && typeof raw.index === 'number') idx = raw.index;
    } catch (e) {}
    if (idx === null) {
      idx = Math.floor(Math.random() * DECK.length);
      try { localStorage.setItem(KEY, JSON.stringify({ date: today, index: idx })); } catch (e) {}
    }
    var it = DECK[idx];
    els.rune.querySelector('.ans-rune-date').textContent = today + ' · 每日一签';
    els.rune.querySelector('.ans-rune-seal').innerHTML =
      '<svg viewBox="0 0 24 24"><path d="' + it[0] + '"/></svg>';
    els.rune.querySelector('.ans-rune-cat').textContent = it[4];
    els.rune.querySelector('.ans-rune-topic').textContent = it[3];
    els.rune.querySelector('.ans-rune-name').textContent = it[1] + ' · ' + it[2];
    els.rune.querySelector('.ans-rune-meaning').textContent = it[5];
    els.rune.querySelector('.ans-rune-guide').textContent = '今日功课：' + it[6];
    startCountdown();
  }

  function openGuide() {
    injectStyles(); buildGuide();
    els.guide.classList.add('is-open');
    els.guide.setAttribute('aria-hidden', 'false');
    document.body.classList.add('ans-ov-open');
  }
  function openRune() {
    injectStyles(); buildRune(); drawRune();
    els.rune.classList.add('is-open');
    els.rune.setAttribute('aria-hidden', 'false');
    document.body.classList.add('ans-ov-open');
  }
  function closeAll() {
    [els.guide, els.rune].forEach(function (ov) {
      if (!ov) return;
      ov.classList.remove('is-open');
      ov.setAttribute('aria-hidden', 'true');
    });
    document.body.classList.remove('ans-ov-open');
    if (fxHide) fxHide();
    if (timer) { clearInterval(timer); timer = null; }
  }

  document.addEventListener('click', function (e) {
    if (e.target.closest('.' + TG)) { e.preventDefault(); openGuide(); return; }
    if (e.target.closest('.' + TR)) { e.preventDefault(); openRune(); return; }
    if (e.target.closest('.ans-ov-backdrop')) { e.preventDefault(); closeAll(); }
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') closeAll();
  });
})();
