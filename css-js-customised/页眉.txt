<script>
(function () {
  // 兜底字体：Jost（Futura 风格的免费替代），有 Futura PT 时会优先用它
  (function () {
    var link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = 'https://fonts.googleapis.com/css2?family=Jost:wght@500&display=swap';
    document.head.appendChild(link);
  })();

  function build() {
    var a = document.querySelector('a.navbar-brand');
    if (!a || a.getAttribute('data-tl-brand')) return;
    var span = a.querySelector('span');
    if (!span) return;
    a.setAttribute('data-tl-brand', '1');

    // 图标 + 文字组 并排居中
    span.style.display = 'flex';
    span.style.alignItems = 'center';
    span.style.gap = '8px';

    span.innerHTML =
      '<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="#8b5cf6" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" style="flex-shrink:0">' +
        '<path d="M12 12V4a1 1 0 0 1 1-1h6.297a1 1 0 0 1 .651 1.759l-4.696 4.025"/>' +
        '<path d="m12 21-7.414-7.414A2 2 0 0 1 4 12.172V6.415a1.002 1.002 0 0 1 1.707-.707L20 20.009"/>' +
        '<path d="m12.214 3.381 8.414 14.966a1 1 0 0 1-.167 1.199l-1.168 1.163a1 1 0 0 1-.706.291H6.351a1 1 0 0 1-.625-.219L3.25 18.8a1 1 0 0 1 .631-1.781l4.165.027"/>' +
      '</svg>' +
      '<span style="position:relative;display:inline-block;line-height:1;transform:translateY(-3px)">' +
        '<span style="font-family:\'Futura PT\',Futura,Jost,\'Noto Sans SC\',sans-serif;font-weight:500;font-size:19px;color:#f1f5f9;letter-spacing:.4px;white-space:nowrap">TechLake</span>' +
        '<span style="position:absolute;right:-14px;bottom:-7px;font-family:Jost,\'Futura PT\',\'Noto Sans SC\',sans-serif;font-size:9px;font-weight:500;color:#a5b4fc;letter-spacing:1.5px;white-space:nowrap">community</span>' +
      '</span>';
  }
  function start() {
    build();
    if (document.body) {
      new MutationObserver(build).observe(document.body, { childList: true, subtree: true });
    }
  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', start);
  } else {
    start();
  }
})();
</script>
